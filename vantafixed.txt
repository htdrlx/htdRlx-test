function Util.closest_player()

    for _, character in workspace.Alive:GetChildren() do
        if character.Name ~= LocalPlayer.Name then
            if character.PrimaryPart then
                local dist = LocalPlayer:DistanceFromCharacter(character.PrimaryPart.Position)

function Util.getCurveCFrame()
    local camera = workspace.CurrentCamera
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return camera.CFrame end
    local method = _G.config.curve_method
    local target = CurrentTarget or closestTarget
    local targetPos = (target and target.PrimaryPart and target.PrimaryPart.Position)
        or (hrp.Position + camera.CFrame.LookVector * 1000)
    local toTarget = (targetPos - hrp.Position).Unit
    if method == "dot" then
        return CFrame.lookAt(hrp.Position, targetPos + Vector3.new(0, 1.75, 0))
    elseif method == "backwards" then
        return CFrame.new(hrp.Position, hrp.Position + (-toTarget) * 1000)
    elseif method == "slow" then
        return CFrame.new(hrp.Position, hrp.Position + Vector3.new(0, -350, 0))
    elseif method == "random" then
        return CFrame.new(hrp.Position, targetPos + Vector3.new(
            math.random(-1000, 1000),
            math.random(-350, 1000),
            math.random(-1000, 1000)
        ))
    else
        return camera.CFrame
    end
end

function Util.ball_curved()
    local ballProps = Util.ball.properties
    local ball = Util.get_ball()
    if not ball then return false end
    local zoomies = ball:FindFirstChild("zoomies")  
    if not zoomies then return false end
    local velocity       = zoomies.VectorVelocity
    local velocityDir    = velocity.Unit
    local toPlayer       = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Unit
    local dotProduct     = toPlayer:Dot(velocityDir)
    local speed          = velocity.Magnitude
    local speedBonus     = math.min(speed / 100, 40)
    local deviationDir   = (velocityDir - velocity).Unit
    local deviationDot   = toPlayer:Dot(deviationDir)
    local curveFactor    = dotProduct - deviationDot
    local distToBall     = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Magnitude
    local pingMs         = Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    local pingThreshold  = 0.5 - (pingMs / 1000)
    local timeToArrival  = distToBall / speed - (pingMs / 1000)
    local parryRadius    =25 - math.min(distToBall / 1000, 25) + speedBonus
    local clampedDot   = math.clamp(dotProduct, -1, 1)
    local targetRad    = math.rad(math.asin(clampedDot))
    ballProps.lerp_radians = linearPredict(ballProps.lerp_radians, targetRad, 0.8)

    if speed > 100 and timeToArrival > pingMs / 10 then
        parryRadius = math.max(parryRadius - 15, 15)
    end
    if distToBall < parryRadius then
        return false
    end
    if curveFactor < pingThreshold then
        return true
    end
    if ballProps.lerp_radians < 0.018 then
        ballProps.last_warping = tick()
    end
    if (tick() - ballProps.last_warping) < (timeToArrival / 1.5) then
        return true
    end
    if (tick() - ballProps.curving) < (timeToArrival / 1.5) then
        return true
    end
    return dotProduct < pingThreshold
end
function Util.get_ball_properties()

    local velocity  = Vector3.zero
    local toPlayer  = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Unit
    local distance  = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Magnitude
    local dot       = toPlayer:Dot(velocity.Unit)
    return { Velocity = velocity, Direction = toPlayer, Distance = distance, Dot = dot }
end

    return { velocity = velocity, direction = direction, distance = distance }
end

local spamRadius = 0
function Util.perfom_spam(params)
    local ballProps = Util.ball.properties
    local ball = Util.get_ball()
    local target = Util.closest_player()
    if not ball then return false end
    if not target or not target.PrimaryPart then return false end
    local velocity     = ball.AssemblyLinearVelocity
    local speed        = velocity.Magnitude
    local toPlayer     = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Unit
    local dot          = toPlayer:Dot(velocity.Unit)
    local targetPos    = target.PrimaryPart.Position
    local distToTarget = LocalPlayer:DistanceFromCharacter(targetPos)
   
    local baseRadius = params.Ping + math.min(speed / 6, 6)

    if params.Entity_Properties.distance > baseRadius then return spamRadius end
    if params.Ball_Properties.Distance   > baseRadius then return spamRadius end
    if distToTarget                      > baseRadius then return spamRadius end

    local dotBonus    = 5 - math.min(speed / 5, 5)
    local dotPenalty  = math.clamp(dot, -1, 0) * dotBonus
    spamRadius        = baseRadius - dotPenalty
    return spamRadius
end

local triggerActive = false
local function triggerParry(ball)
    if triggerActive then return end
    triggerActive = true
    Util.parry()

    ball:GetAttributeChangedSignal("target"):Once(function()
        triggerActive = false
    end)

    local startTime = tick()
    task.spawn(function()
        repeat
            RunService.PreSimulation:Wait()
        until (tick() - startTime >= 1 or not triggerActive)
        triggerActive = false
    end)
end

local antiPhantomActive = false
MainTab:AddToggle("Toggle", {
    Title   = "anti phantom",
    Default = _G.config.anti_phantom or false,
    Callback = function(enabled)
        _G.config.anti_phantom = enabled
        if enabled then
            activeConnections.anti_phantom = workspace.Runtime.ChildAdded:Connect(function(part)
                local character = LocalPlayer.Character
                if not character then return end
                if not (part.Name:lower() == "maxtransmission" or part.Name == "transmissionpart") then return end
                local weld = part:FindFirstChildWhichIsA("WeldConstraint")
                if not weld then return end
                for _, ball in workspace.Balls:GetChildren() do
                    if not ball:FindFirstChild("zoomies") then continue end
                    if antiPhantomActive then continue end
                    antiPhantomActive = true
                    local cancelled = false
                    task.spawn(function()
                        local startTime = tick()
         
                        while tick() - startTime < 1 and ball.Parent and not cancelled do
                            task.wait()
                            game:GetService("TweenService"):Create(
                                character.HumanoidRootPart,
                                TweenInfo.new(0.1, Enum.EasingStyle.Linear, Enum.EasingDirection.Out),
                                { CFrame = ball.CFrame * CFrame.new(0, 0, -3) }
                            ):Play()
                        end
                        antiPhantomActive = false
                    end)
         
                    ball:GetAttributeChangedSignal("target"):Once(function()
                        cancelled = true
                        antiPhantomActive = false
                    end)
                end
            end)
        else
            if activeConnections.anti_phantom then
                activeConnections.anti_phantom:Disconnect()
                activeConnections.anti_phantom = nil
            end
            antiPhantomActive = false
        end
    end
})

            activeConnections.auto_spam = RunService.PreSimulation:Connect(function()
                local ball = Util.get_ball()
                if not ball then return end
                local zoomies = ball:FindFirstChild("zoomies")
                if not zoomies then return end
                Util.closest_player()
                local pingMs        = Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
                local pingAdjusted  = math.clamp(pingMs / 10, 1, 16)
                local ballTarget    = ball:GetAttribute("target")
                local ballProperties   = Util:get_ball_properties()
                local playerProperties = Util:players_properties()
                local spamDist = Util.perfom_spam({
                    Ball_Properties   = ballProperties,
                    Entity_Properties = playerProperties,
                    Ping              = pingAdjusted,
                })
                local targetPos = closestTarget.PrimaryPart.Position
                if not targetPos then return end
                local distToTarget    = LocalPlayer:DistanceFromCharacter(targetPos)
                local toBall          = (LocalPlayer.Character.PrimaryPart.Position - ball.Position).Unit
                local ballVelUnit     = zoomies.VectorVelocity.Unit
                toBall:Dot(ballVelUnit)  
                local distToBall      = LocalPlayer:DistanceFromCharacter(ball.Position)
                if not ballTarget then return end
               
                if distToTarget > spamDist or distToBall > spamDist then return end
           
                local isPulsed = LocalPlayer.Character:GetAttribute("Pulsed")
                if isPulsed then return end
           
                if ballTarget == LocalPlayer.Name and distToTarget > 30 and distToBall > 30 then return end
             
                local threshold = _G.config.spam_threshold
                if distToBall <= spamDist and parryCount > threshold then
                    Util.parry()
                end
            end)
        else
            if activeConnections.auto_spam then
                activeConnections.auto_spam:Disconnect()
                activeConnections.auto_spam = nil
            end
        end

local korbloxRespawnConn
local korbloxHeadlessConnection

local function updateKorbloxHeadless()
    local character = LocalPlayer.Character
    if not character then return end
    
    local head = character:FindFirstChild("Head")
    if head then
        if head.Transparency ~= 1 then
            head.Transparency = 1
        end
        local face = head:FindFirstChild("face")
        if face and face:IsA("Decal") then
            if face.Transparency ~= 1 then
                face.Transparency = 1
            end
        end
    end
    
    local r15Parts = {"RightUpperLeg", "RightLowerLeg", "RightFoot"}
    for _, partName in ipairs(r15Parts) do
        local part = character:FindFirstChild(partName)
        if part and part:IsA("BasePart") then
            if part.Transparency ~= 1 then
                part.Transparency = 1
            end
        end
    end
    
    local r6Part = character:FindFirstChild("Right Leg")
    if r6Part and r6Part:IsA("BasePart") then
        if r6Part.Transparency ~= 1 then
            r6Part.Transparency = 1
        end
    end
end

local function restoreKorbloxHeadless()
    local character = LocalPlayer.Character
    if not character then return end
    
    local head = character:FindFirstChild("Head")
    if head then
        head.Transparency = 0
        local face = head:FindFirstChild("face")
        if face and face:IsA("Decal") then
            face.Transparency = 0
        end
    end
    
    local r15Parts = {"RightUpperLeg", "RightLowerLeg", "RightFoot"}
    for _, partName in ipairs(r15Parts) do
        local part = character:FindFirstChild(partName)
        if part and part:IsA("BasePart") then
            part.Transparency = 0
        end
    end
    
    local r6Part = character:FindFirstChild("Right Leg")
    if r6Part and r6Part:IsA("BasePart") then
        r6Part.Transparency = 0
    end
end

local function applyKorbloxHeadless(enable)
    if enable then
        if not korbloxHeadlessConnection then
            korbloxHeadlessConnection = RunService.RenderStepped:Connect(updateKorbloxHeadless)
        end
    else
        if korbloxHeadlessConnection then
            korbloxHeadlessConnection:Disconnect()
            korbloxHeadlessConnection = nil
        end
        restoreKorbloxHeadless()
    end
end
local function startKorbloxLoop()
    if korbloxRespawnConn then korbloxRespawnConn:Disconnect(); korbloxRespawnConn = nil end
    korbloxRespawnConn = LocalPlayer.CharacterAdded:Connect(function(char)
        local humanoid = char:WaitForChild("Humanoid", 5)
        if not humanoid then return end
        task.wait(1)
        applyKorbloxHeadless(true)
    end)
end
local function stopKorbloxLoop()
    if korbloxRespawnConn then korbloxRespawnConn:Disconnect(); korbloxRespawnConn = nil end
end
VisualTab:AddToggle("KorbloxHeadless", {
    Title   = "korblox + headless",
    Default = _G.config.korblox_headless or false,
    Callback = function(enabled)
        _G.config.korblox_headless = enabled
        task.spawn(function()
            if enabled then
                applyKorbloxHeadless(true)
                startKorbloxLoop()
            else
                stopKorbloxLoop()
                applyKorbloxHeadless(false)
            end
        end)
    end
})

        end
