local function FormatTime(seconds)
    seconds = math.max(0, math.floor(seconds))
    return string.format("%02d:%02d:%02d",
        math.floor(seconds / 3600),
        math.floor((seconds % 3600) / 60),
        seconds % 60)
end

local function GetCharacter()
    local character = player.Character
    if not character then return end
    local hrp      = character:FindFirstChild("HumanoidRootPart")
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not hrp or not humanoid or humanoid.Health <= 0 then return end
    return character, hrp, humanoid
end

local function IsFruitObject(obj)
    if not obj then return false end
    -- Chá»‰ nháº­n Tool hoáº·c Model/Part trĂ¹ng tĂªn chĂ­nh xĂ¡c, KHĂ”NG dĂ¹ng :find("fruit")
    -- vĂ¬ "fruit" cĂ³ thá»ƒ match tĂªn NPC hoáº·c GUI element
    return FruitNames[obj.Name] == true
end

-- âœ… FIX NPC: Kiá»ƒm tra object cĂ³ náº±m trong model chá»©a Humanoid khĂ´ng
-- (NPC, player character, Backpack Ä‘á»u bá»‹ loáº¡i bá»)
local function IsInsideAnyCharacter(obj)
    local p = obj.Parent
    while p and p ~= workspace do
        if p:IsA("Backpack") then return true end          -- trong tĂºi ai Ä‘Ă³
        if p:FindFirstChildOfClass("Humanoid") then        -- trong NPC / character
            return true
        end
        p = p.Parent
    end
    return false
end

local function FindFruit()
    local _, hrp = GetCharacter()
    if not hrp then return end
    local closest, closestDist = nil, math.huge
    for _, obj in ipairs(workspace:GetDescendants()) do
        -- âœ… FIX NPC: bá» qua náº¿u náº±m trong character/NPC/Backpack
        if IsFruitObject(obj) and not IsInsideAnyCharacter(obj) then
            local part
            if obj:IsA("Tool") then
                part = obj:FindFirstChild("Handle")
            elseif obj:IsA("BasePart") then
                part = obj
            else
                part = obj:FindFirstChildWhichIsA("BasePart", true)
            end
            if part then
                local d = (hrp.Position - part.Position).Magnitude
                if d < closestDist then
                    closestDist = d
                    closest     = obj
                end
            end
        end
    end
    return closest
end

local function GetFruitPart(fruit)
    if not fruit then return end
    if fruit:IsA("Tool")     then return fruit:FindFirstChild("Handle") end
    if fruit:IsA("BasePart") then return fruit end
    return fruit:FindFirstChildWhichIsA("BasePart", true)
end

-- ================= NHáº¶T TRĂI =================
local function CollectFruit(fruit)
    if not fruit or not fruit.Parent then return false end
    if fruit:IsA("Tool") then
        if fruit.Parent ~= player.Backpack and fruit.Parent ~= player.Character then
            local ok, err = pcall(function() fruit.Parent = player.Backpack end)
            if ok then
                CollectCount += 1
                table.insert(CollectedFruits, fruit.Name)
                print("[XCrzy] Nháº·t: " .. fruit.Name .. " (Tá»•ng: " .. CollectCount .. ")")
                return true
            else
                warn("[XCrzy] Lá»—i nháº·t: " .. tostring(err))
            end
        end
    end
    return false
end

-- ================= TWEEN BAY Äáº¾N TRĂI =================
local CurrentTween = nil

local function StopTween()
    if CurrentTween then
        pcall(function() CurrentTween:Cancel() end)
        CurrentTween = nil
    end
end

local function TweenToFruit(fruit)
    local character, hrp, humanoid = GetCharacter()
    local part = GetFruitPart(fruit)
    if not character or not hrp or not humanoid or not part then return end

    StopTween()

    local targetPos = part.Position + Vector3.new(0, FruitHeight, 0)
    local dist      = (hrp.Position - targetPos).Magnitude

    if dist <= 4 then
        CollectFruit(fruit)
        return
    end

    local duration = math.max(dist / TweenSpeed, 0.05)

    -- âœ… FIX 2: Anchor HRP â†’ cháº·n trá»ng lá»±c, tween khĂ´ng bá»‹ Ä‘Ă¡nh báº¡i bá»Ÿi physics
    hrp.Anchored          = true
    humanoid.AutoRotate   = false

    CurrentTween = TweenService:Create(
        hrp,
        TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut),
        { CFrame = CFrame.new(targetPos) }
    )
    CurrentTween:Play()

    local finished = false
    local conn
    conn = CurrentTween.Completed:Connect(function()
        finished = true
        conn:Disconnect()
    end)

    -- âœ… FIX 1: FruitEnabled giá» Ä‘Ăºng (= true), loop cháº¡y bĂ¬nh thÆ°á»ng
    while not finished and FruitEnabled do
        if not fruit.Parent or not part.Parent then break end
        local d = (hrp.Position - part.Position).Magnitude
        if d <= 6 then
            CollectFruit(fruit)
            break
        end
        RunService.Heartbeat:Wait()
    end

    StopTween()
    pcall(function() if conn then conn:Disconnect() end end)

    -- Thá»­ nháº·t thĂªm 1 láº§n sau khi bay xong (safety net)
    if fruit and fruit.Parent then
        CollectFruit(fruit)
    end

    -- âœ… FIX 2: Unanchor + khĂ´i phá»¥c humanoid
    hrp.Anchored = false
    if humanoid.Parent then
        humanoid.AutoRotate = true
        pcall(function() humanoid:ChangeState(Enum.HumanoidStateType.GettingUp) end)
    end
end

task.spawn(function()
    while FruitEnabled do
        local fruit = FindFruit()
        if fruit then
            CurrentFruit = fruit.Name
            TweenToFruit(fruit)
        else
            CurrentFruit = "KhĂ´ng cĂ³"
            task.wait(0.5)
        end
        task.wait(0.15)
    end
end)

task.spawn(function()
    while task.wait(1) do
        local elapsed   = os.clock() - SessionStart
        local fruit     = FindFruit()
        CurrentFruit    = fruit and fruit.Name or "KhĂ´ng cĂ³"
        local remaining = HopAfter - elapsed

        statusLabel.Text = string.format(
            "Fruit: %s\nServer: %s\nHop in: %s\nCollected: %d",
            CurrentFruit,
            FormatTime(elapsed),
            remaining > 0 and FormatTime(remaining) or "READY",
            CollectCount
        )

        if HopEnabled and elapsed >= HopAfter and not fruit and not HopRequested then
            HopRequested = true
            pcall(function() TeleportService:Teleport(game.PlaceId, player) end)
        end
    end
end)
